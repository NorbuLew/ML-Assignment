# used anywhere in this project; the CPU build is the correct one.
--extra-index-url https://download.pytorch.org/whl/cpu

numpy>=2.0

# Streamlit 1.38+ for the `width="stretch"` chart/dataframe argument; the older

streamlit>=1.38

# Altair 5.5 is a hard floor, not a preference: dashboard/lib/theme.py registers
# the chart theme with the `@alt.theme.register` decorator, which does not exist
# before 5.5. The 5.4 API (`alt.themes.register`) is deprecated and would raise
# AttributeError here.
altair>=5.5

ipykernel>=6.29

# tools/nb_to_pdf.py exports the executed notebooks to PDF for submission. It
# goes notebook -> HTML -> PDF through the Chrome already on the machine, so it
# needs nbconvert but neither a LaTeX install nor a playwright download.
nbconvert>=7.16
nbformat>=5.10
